-- RiverGate Cooperative: SQL examples (generic SQL; adapt syntax to your DBMS)
CREATE TABLE CanalPool (pool_id INT PRIMARY KEY, balance DECIMAL(12,2) NOT NULL CHECK (balance >= 0));
CREATE TABLE FarmDaily (farm_id INT NOT NULL, allocation_date DATE NOT NULL, allocated DECIMAL(12,2) NOT NULL DEFAULT 0, daily_limit DECIMAL(12,2) NOT NULL, PRIMARY KEY (farm_id, allocation_date), CHECK (allocated >= 0), CHECK (daily_limit >= 0));
CREATE TABLE MovementRequest (request_id VARCHAR(80) PRIMARY KEY, farm_id INT NOT NULL, source_pool INT NOT NULL, destination_pool INT NOT NULL, quantity DECIMAL(12,2) NOT NULL CHECK (quantity > 0), allocation_date DATE NOT NULL, status VARCHAR(20) NOT NULL);
CREATE TABLE MovementLog (movement_id INTEGER PRIMARY KEY, request_id VARCHAR(80) NOT NULL UNIQUE, farm_id INT NOT NULL, source_pool INT NOT NULL, destination_pool INT NOT NULL, quantity DECIMAL(12,2) NOT NULL, allocation_date DATE NOT NULL);
