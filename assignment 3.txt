-- =========================================================
-- 2) NORMALISED SCHEMA: 3NF/BCNF core relations
-- =========================================================
DROP TABLE IF EXISTS ORDER_ITEM;
DROP TABLE IF EXISTS ORDERS;
DROP TABLE IF EXISTS CUSTOMER;
DROP TABLE IF EXISTS ADDRESS;
DROP TABLE IF EXISTS PRODUCT;
DROP TABLE IF EXISTS CATEGORY;
DROP TABLE IF EXISTS SUPPLIER;
DROP TABLE IF EXISTS WAREHOUSE;
DROP TABLE IF EXISTS PIN_LOCATION;

CREATE TABLE CUSTOMER (
    CustomerID INT PRIMARY KEY,
    CustomerName VARCHAR(100) NOT NULL,
    CustomerEmail VARCHAR(150) NOT NULL UNIQUE
);

CREATE TABLE PIN_LOCATION (
    ShipPIN VARCHAR(10) PRIMARY KEY,
    ShipCity VARCHAR(100) NOT NULL,
    ShipState VARCHAR(100) NOT NULL
);

CREATE TABLE ADDRESS (
    AddressID INT PRIMARY KEY,
    ShipStreet VARCHAR(150) NOT NULL,
    ShipCity VARCHAR(100) NOT NULL,
    ShipPIN VARCHAR(10) NOT NULL,
    FOREIGN KEY (ShipPIN) REFERENCES PIN_LOCATION(ShipPIN)
);

CREATE TABLE CATEGORY (
    CategoryID INT PRIMARY KEY,
    CategoryName VARCHAR(100) NOT NULL UNIQUE
);

CREATE TABLE SUPPLIER (
    SupplierID INT PRIMARY KEY,
    SupplierName VARCHAR(100) NOT NULL,
    SupplierPhone VARCHAR(20) NOT NULL
);

CREATE TABLE PRODUCT (
    ProductID INT PRIMARY KEY,
    ProductName VARCHAR(150) NOT NULL,
    UnitPrice DECIMAL(10,2) NOT NULL CHECK (UnitPrice >= 0),
    CategoryID INT NOT NULL,
    SupplierID INT NOT NULL,
    FOREIGN KEY (CategoryID) REFERENCES CATEGORY(CategoryID),
    FOREIGN KEY (SupplierID) REFERENCES SUPPLIER(SupplierID)
);

CREATE TABLE WAREHOUSE (
    WarehouseID INT PRIMARY KEY,
    WarehouseCity VARCHAR(100) NOT NULL
);

CREATE TABLE ORDERS (
    OrderID INT PRIMARY KEY,
    OrderDate DATE NOT NULL,
    CustomerID INT NOT NULL,
    AddressID INT NOT NULL,
    FOREIGN KEY (CustomerID) REFERENCES CUSTOMER(CustomerID),
    FOREIGN KEY (AddressID) REFERENCES ADDRESS(AddressID)
);

CREATE TABLE ORDER_ITEM (
    OrderID INT NOT NULL,
    ProductID INT NOT NULL,
    Quantity INT NOT NULL CHECK (Quantity > 0),
    WarehouseID INT NOT NULL,
    PRIMARY KEY (OrderID, ProductID),
    FOREIGN KEY (OrderID) REFERENCES ORDERS(OrderID),
    FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID),
    FOREIGN KEY (WarehouseID) REFERENCES WAREHOUSE(WarehouseID)
);

-- =========================================================
-- 3) 4NF: independent tags and colours (no tag-colour cross product)
-- =========================================================
DROP TABLE IF EXISTS PRODUCT_COLOUR;
DROP TABLE IF EXISTS PRODUCT_TAG;

CREATE TABLE PRODUCT_TAG (
    ProductID INT NOT NULL,
    Tag VARCHAR(80) NOT NULL,
    PRIMARY KEY (ProductID, Tag),
    FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID)
);

CREATE TABLE PRODUCT_COLOUR (
    ProductID INT NOT NULL,
    Colour VARCHAR(80) NOT NULL,
    PRIMARY KEY (ProductID, Colour),
    FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID)
);

-- BCNF but not 4NF comparison relation: every tag-colour combination
DROP TABLE IF EXISTS PRODUCT_TAG_COLOUR;
CREATE TABLE PRODUCT_TAG_COLOUR (
    ProductID INT NOT NULL,
    Tag VARCHAR(80) NOT NULL,
    Colour VARCHAR(80) NOT NULL,
    PRIMARY KEY (ProductID, Tag, Colour),
    FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID)
);

-- =========================================================
-- 4) 5NF supply relation and its projections
-- =========================================================
DROP TABLE IF EXISTS SUPPLIER_CATEGORY_WAREHOUSE;
DROP TABLE IF EXISTS SUPPLIER_WAREHOUSE;
DROP TABLE IF EXISTS CATEGORY_WAREHOUSE;
DROP TABLE IF EXISTS SUPPLIER_CATEGORY;

CREATE TABLE SUPPLIER_CATEGORY (
    SupplierID INT NOT NULL,
    CategoryID INT NOT NULL,
    PRIMARY KEY (SupplierID, CategoryID),
    FOREIGN KEY (SupplierID) REFERENCES SUPPLIER(SupplierID),
    FOREIGN KEY (CategoryID) REFERENCES CATEGORY(CategoryID)
);

CREATE TABLE CATEGORY_WAREHOUSE (
    CategoryID INT NOT NULL,
    WarehouseID INT NOT NULL,
    PRIMARY KEY (CategoryID, WarehouseID),
    FOREIGN KEY (CategoryID) REFERENCES CATEGORY(CategoryID),
    FOREIGN KEY (WarehouseID) REFERENCES WAREHOUSE(WarehouseID)
);

CREATE TABLE SUPPLIER_WAREHOUSE (
    SupplierID INT NOT NULL,
    WarehouseID INT NOT NULL,
    PRIMARY KEY (SupplierID, WarehouseID),
    FOREIGN KEY (SupplierID) REFERENCES SUPPLIER(SupplierID),
    FOREIGN KEY (WarehouseID) REFERENCES WAREHOUSE(WarehouseID)
);

-- 5) DESIGN CHALLENGE: store historical paid price per item
-- =========================================================
-- Add PricePaid to ORDER_ITEM so old orders retain their actual price.
ALTER TABLE ORDER_ITEM ADD COLUMN PricePaid DECIMAL(10,2) NOT NULL DEFAULT 0
    CHECK (PricePaid >= 0);
UPDATE ORDER_ITEM oi JOIN PRODUCT p ON oi.ProductID=p.ProductID
SET oi.PricePaid=p.UnitPrice WHERE oi.PricePaid=0;

-- Price history, optional alternative for product price changes:
DROP TABLE IF EXISTS PRODUCT_PRICE_HISTORY;
CREATE TABLE PRODUCT_PRICE_HISTORY (
    ProductID INT NOT NULL,
    EffectiveFrom DATETIME NOT NULL,
    EffectiveTo DATETIME NULL,
    Price DECIMAL(10,2) NOT NULL CHECK (Price >= 0),
    PRIMARY KEY (ProductID, EffectiveFrom),
    FOREIGN KEY (ProductID) REFERENCES PRODUCT(ProductID),
    CHECK (EffectiveTo IS NULL OR EffectiveTo > EffectiveFrom)
);

-- =========================================================
-- 6) Useful query examples
-- =========================================================
-- Order history with customer, product, supplier, category and warehouse:
SELECT o.OrderID, o.OrderDate, c.CustomerName, p.ProductName,
       oi.Quantity, oi.PricePaid, s.SupplierName, cat.CategoryName,
       w.WarehouseCity
FROM ORDERS o
JOIN CUSTOMER c ON c.CustomerID=o.CustomerID
JOIN ORDER_ITEM oi ON oi.OrderID=o.OrderID
JOIN PRODUCT p ON p.ProductID=oi.ProductID
JOIN SUPPLIER s ON s.SupplierID=p.SupplierID
JOIN CATEGORY cat ON cat.CategoryID=p.CategoryID
JOIN WAREHOUSE w ON w.WarehouseID=oi.WarehouseID
ORDER BY o.OrderDate DESC, o.OrderID;

-- Customer order history:
SELECT o.OrderID, o.OrderDate, p.ProductName, oi.Quantity, oi.PricePaid
FROM ORDERS o
JOIN ORDER_ITEM oi ON oi.OrderID=o.OrderID
JOIN PRODUCT p ON p.ProductID=oi.ProductID
WHERE o.CustomerID=1
ORDER BY o.OrderDate DESC;

-- Supplier phone update (one fact, one row):
UPDATE SUPPLIER SET SupplierPhone='9876500099' WHERE SupplierID=201;

-- Product current price update:
UPDATE PRODUCT SET UnitPrice=1250.00 WHERE ProductID=501;

-- Tags for a product:
SELECT Tag FROM PRODUCT_TAG WHERE ProductID=501;
-- Colours for a product:
SELECT Colour FROM PRODUCT_COLOUR WHERE ProductID=501;

-- Reconstruct independent tag/colour combinations (not usually needed):
SELECT t.ProductID, t.Tag, c.Colour
FROM PRODUCT_TAG t JOIN PRODUCT_COLOUR c ON c.ProductID=t.ProductID
WHERE t.ProductID=501;
