DROP DATABASE IF EXISTS swiftspan;
CREATE DATABASE swiftspan;
USE swiftspan;

CREATE TABLE Sender(
 SenderID INT PRIMARY KEY,
 SenderName VARCHAR(80) NOT NULL,
 RouteGroup VARCHAR(30) NOT NULL
);
CREATE TABLE Service(
 ServiceID INT PRIMARY KEY,
 ServiceCode VARCHAR(20) NOT NULL UNIQUE,
 ServiceName VARCHAR(80) NOT NULL,
 VolumeWeight DECIMAL(8,2) NOT NULL CHECK (VolumeWeight >= 0)
);
CREATE TABLE Coordinator(
 CoordinatorID INT PRIMARY KEY,
 CoordinatorName VARCHAR(80) NOT NULL,
 Hub VARCHAR(50) NOT NULL
);
CREATE TABLE Assignment(
 SenderID INT NOT NULL,
 ServiceID INT NOT NULL,
 CoordinatorID INT NOT NULL,
 PRIMARY KEY(SenderID,ServiceID),
 FOREIGN KEY(SenderID) REFERENCES Sender(SenderID) ON DELETE RESTRICT,
 FOREIGN KEY(ServiceID) REFERENCES Service(ServiceID) ON DELETE RESTRICT,
 FOREIGN KEY(CoordinatorID) REFERENCES Coordinator(CoordinatorID) ON DELETE RESTRICT
);
CREATE TABLE PolicyBand(
 Completion INT PRIMARY KEY CHECK (Completion BETWEEN 0 AND 100),
 ServiceBand VARCHAR(20) NOT NULL,
 RouteFactor DECIMAL(6,2) NOT NULL
);
CREATE TABLE Observation(
 SenderID INT NOT NULL,
 ServiceID INT NOT NULL,
 DispatchWave INT NOT NULL,
 Completion INT NOT NULL CHECK (Completion BETWEEN 0 AND 100),
 PRIMARY KEY(SenderID,ServiceID,DispatchWave),
 FOREIGN KEY(SenderID,ServiceID) REFERENCES Assignment(SenderID,ServiceID) ON DELETE RESTRICT,
 FOREIGN KEY(Completion) REFERENCES PolicyBand(Completion) ON DELETE RESTRICT
);

-- Q1: Actor 1, wave 4, weighted adjusted index = Completion * VolumeWeight * RouteFactor
SELECT o.SenderID,o.ServiceID,s.ServiceName,o.DispatchWave,o.Completion,
       p.RouteFactor,
       o.Completion*s.VolumeWeight*p.RouteFactor AS WeightedAdjustedIndex
FROM Observation o JOIN Service s USING(ServiceID)
JOIN PolicyBand p ON p.Completion=o.Completion
WHERE o.SenderID=1 AND o.DispatchWave=4
ORDER BY o.ServiceID;

-- Q2: Senders whose every required service has at least one observation >= 39
SELECT se.SenderID,se.SenderName
FROM Sender se
WHERE NOT EXISTS (
 SELECT 1 FROM Service sv
 WHERE NOT EXISTS (
  SELECT 1 FROM Observation o
  WHERE o.SenderID=se.SenderID AND o.ServiceID=sv.ServiceID
    AND o.Completion>=39
 )
);

-- Q3: Observed services with no completion below 39 in wave 4
SELECT se.SenderID,se.SenderName,sv.ServiceID,sv.ServiceName
FROM Observation o JOIN Sender se USING(SenderID)
JOIN Service sv USING(ServiceID)
WHERE o.DispatchWave=4
GROUP BY se.SenderID,se.SenderName,sv.ServiceID,sv.ServiceName
HAVING MIN(o.Completion)>=39;

-- Q4: Coordinator means below observation-weighted hub mean
WITH coord_mean AS (
 SELECT c.CoordinatorID,c.CoordinatorName,c.Hub,AVG(o.Completion) AS avg_completion
 FROM Observation o JOIN Assignment a USING(SenderID,ServiceID)
 JOIN Coordinator c USING(CoordinatorID)
 GROUP BY c.CoordinatorID,c.CoordinatorName,c.Hub
), hub_mean AS (
 SELECT c.Hub,AVG(o.Completion) AS hub_avg
 FROM Observation o JOIN Assignment a USING(SenderID,ServiceID)
 JOIN Coordinator c USING(CoordinatorID)
 GROUP BY c.Hub
)
SELECT cm.CoordinatorID,cm.CoordinatorName,cm.Hub,cm.avg_completion,hm.hub_avg
FROM coord_mean cm JOIN hub_mean hm USING(Hub)
WHERE cm.avg_completion<hm.hub_avg;

-- Q5: Senders with sub-threshold observations in >=3 distinct waves
SELECT se.SenderID,se.SenderName,COUNT(DISTINCT o.DispatchWave) AS low_waves
FROM Sender se JOIN Observation o USING(SenderID)
WHERE o.Completion<39
GROUP BY se.SenderID,se.SenderName
HAVING COUNT(DISTINCT o.DispatchWave)>=3;

-- Purpose-specific views
CREATE VIEW ActorObservationView AS
SELECT se.SenderID,se.SenderName,se.RouteGroup,o.ServiceID,sv.ServiceName,
       o.DispatchWave,o.Completion
FROM Sender se JOIN Observation o USING(SenderID)
JOIN Service sv USING(ServiceID);

CREATE VIEW OperationalHandoverView AS
SELECT o.SenderID,o.ServiceID,o.DispatchWave,o.Completion,
       a.CoordinatorID,c.CoordinatorName,c.Hub,sv.ServiceCode,sv.ServiceName,
       sv.VolumeWeight,p.ServiceBand,p.RouteFactor
FROM Observation o JOIN Assignment a USING(SenderID,ServiceID)
JOIN Coordinator c USING(CoordinatorID)
JOIN Service sv USING(ServiceID)
JOIN PolicyBand p ON p.Completion=o.Completion;
